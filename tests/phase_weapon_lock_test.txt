# A locked weapon stays on (host/addons/phase.c).  The TP set does not always name a weapon,
# so filtering a pinned slot out of a set is not enough: the exact copy is captured, a
# weaponskill keeps it instead of swapping it for sets.WS.<name>, and it is re-armed when the
# weapon is drawn.  A cast while disengaged may swap the hand for the cast and put it back; a
# Bard's Nitro song is the only exception while engaged.  Run by
# tests/phase_weapon_lock_test.sh.
frames 2
lua control xi.phase.set{enabled=true, player=16909060, zone=230, town=1, job=5, sets={{{1,0,0},{2,1,0},{3,4,0},{4,2,0},{5,3,0}},{{10,0,0},{11,1,0},{12,4,0}},{{20,0,0},{21,1,0},{22,4,0}}}, names={'idle','ws','magic'}, engage=1, idle=1, town=1, rules={{category=7, action=78, pre=2},{category=3, action=5, pre=3, mid=3},{category=3, action=100, pre=3, mid=3}}} return xi.phase.status().sets
